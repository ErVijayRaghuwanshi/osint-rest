# 1. Create the Security Group
ALB_SG_ID=$(aws ec2 create-security-group \
    --group-name nautilus-sg \
    --description "Public HTTP access for ALB" \
    --query 'GroupId' --output text)

# 2. Allow Inbound Port 80 from anywhere
aws ec2 authorize-security-group-ingress \
    --group-id $ALB_SG_ID \
    --protocol tcp \
    --port 80 \
    --cidr 0.0.0.0/0






# 1. Create the Target Group (Targeting instances on port 80)
TG_ARN=$(aws ec2 create-target-group \
    --name nautilus-tg \
    --protocol HTTP \
    --port 80 \
    --vpc-id vpc-0bf2726a05e4f82b6 \
    --target-type instance \
    --query 'TargetGroups[0].TargetGroupArn' --output text)

# 2. Register the nautilus-ec2 instance
# (Replace INSTANCE_ID with the ID of nautilus-ec2)
aws ec2 register-targets \
    --target-group-arn $TG_ARN \
    --targets Id=i-08ffe884d7b41c3f6

# Create the ALB
ALB_ARN=$(aws elbv2 create-load-balancer \
    --name nautilus-alb \
    --subnets <SUBNET_1_ID> <SUBNET_2_ID> \
    --security-groups $ALB_SG_ID \
    --query 'LoadBalancers[0].LoadBalancerArn' --output text)

# Create a Listener to forward traffic from ALB port 80 to the Target Group
aws elbv2 create-listener \
    --load-balancer-arn $ALB_ARN \
    --protocol HTTP \
    --port 80 \
    --default-actions Type=forward,TargetGroupArn=$TG_ARN


# Get the Security Group ID of the EC2 instance
EC2_SG_ID=$(aws ec2 describe-instances --filters "Name=tag:Name,Values=nautilus-ec2" --query "Reservations[*].Instances[*].SecurityGroups[0].GroupId" --output text)

# Allow traffic ONLY from the ALB Security Group
aws ec2 authorize-security-group-ingress \
    --group-id $EC2_SG_ID \
    --protocol tcp \
    --port 80 \
    --source-group $ALB_SG_ID
